\begin{notes}\small
\textbf{Wiederholung und Schreibweisen}
\begin{itemize}
  \item Eine \textbf{Grammatik} ist ein 4-Tupel $G=(V,\Sigma,P,S)$ aus Variablen $V$, Terminalsymbolen $\Sigma$, Regeln $P$ und Startsymbol $S\in V$. Regeln mit derselben linken Seite fassen wir zusammen: $S\to aSb\mid\varepsilon$ steht für die beiden Regeln $S\to aSb$ und $S\to\varepsilon$.
  \item Wenn nichts anderes gesagt wird, sind Großbuchstaben Variablen, $S$ ist das Startsymbol und alle übrigen Zeichen sind Terminalsymbole. Es genügt dann, die Regeln anzugeben.
  \item $u\Rightarrow_G v$ heißt: $v$ entsteht aus $u$ durch \emph{eine} Regelanwendung. $\Rightarrow^*_G$ bedeutet beliebig viele (auch null) Schritte. Es ist $L(G)=\{w\in\Sigma^*\mid S\Rightarrow^*_G w\}$. Alles, was aus $S$ ableitbar ist (auch mit Variablen darin), nennen wir eine \textbf{Satzform}. $L(G)$ besteht also genau aus den Satzformen ohne Variablen.
  \item Für ein Wort $w$ ist $|w|$ seine Länge, $|w|_x$ die Anzahl der Vorkommen des Zeichens $x$ in $w$ und $w^R$ das Spiegelwort (z.\,B. $(aab)^R=baa$). $a^n$ ist das Wort aus $n$ Zeichen $a$, insbesondere ist $a^0=\varepsilon$.
  \item \textbf{Chomsky-Hierarchie:} $G$ ist vom Typ~0, wenn es keine Einschränkungen gibt, vom Typ~1, wenn für alle Regeln $u\to v$ gilt $|u|\leq|v|$, vom Typ~2, wenn jede linke Seite eine einzelne Variable ist ($u\in V$), und vom Typ~3, wenn zusätzlich jede rechte Seite in $\Sigma\cup\Sigma V$ liegt. Die $\varepsilon$-Sonderregel $S\to\varepsilon$ ist erlaubt (bei Typ~1 soll $S$ dann auf keiner rechten Seite vorkommen). \emph{Konvention für dieses Blatt:} Bei Typ~2 und Typ~3 lassen wir, wie allgemein üblich, beliebige Regeln $A\to\varepsilon$ zu. Man kann zeigen, dass sich die erzeugbaren Sprachen dadurch nicht ändern.
  \item Eine \emph{Sprache} ist vom Typ~$i$, wenn es \emph{irgendeine} Typ-$i$-Grammatik gibt, die sie erzeugt.
  \item \textbf{Wie zeigt man $L(G)=L$?} Man zeigt beide Inklusionen.
  \begin{itemize}
    \item[$\supseteq$] Jedes Wort aus $L$ ist ableitbar: Für ein beliebiges $w\in L$ gibt man eine Ableitung an, meist als Schema (\enquote{$n$-mal Regel 1, dann einmal Regel 2}).
    \item[$\subseteq$] $G$ erzeugt nichts Falsches: Man sucht eine Eigenschaft (\emph{Invariante}), die $S$ hat und die bei jeder Regelanwendung erhalten bleibt. Dann haben alle Satzformen diese Eigenschaft (Induktion über die Länge der Ableitung), und für Satzformen ohne Variablen muss daraus $w\in L$ folgen.
  \end{itemize}
\end{itemize}
\end{notes}

\aufgabetitel{$6$}{Ableitungen}\\
Gegeben sei die Grammatik $G=(\{S\},\{a,b\},P,S)$ mit
\[P:\qquad S\to aSb\mid bSa\mid\varepsilon.\]
\begin{enumerate}[a)]
  \item Geben Sie Ableitungen für die Wörter $aabb$, $abab$ und $bbaa$ an.
  \item Bestimmen Sie $L_4:=\{w\in L(G)\mid |w|\leq 4\}$. Gehören $abba$ und $baab$ zu $L(G)$? Begründen Sie.
  \item Zeigen Sie: Für jedes $w\in L(G)$ gilt $|w|_a=|w|_b$.
  \item Nach c) ist $L(G)\subseteq L_{=}:=\{w\in\{a,b\}^*\mid |w|_a=|w|_b\}$, nach b) gilt aber keine Gleichheit. Beschreiben Sie $L(G)$ exakt.\\
  \emph{Hinweis:} Vergleichen Sie den $i$-ten Buchstaben von vorne mit dem $i$-ten Buchstaben von hinten.
  \item \textit{(anspruchsvoller)} Ergänzen Sie $P$ um \emph{eine} Regel, sodass die neue Grammatik $G'$ genau die Sprache $L_{=}$ erzeugt, und begründen Sie Ihre Wahl.\\
  \emph{Hinweis:} Betrachten Sie für $w\in L_{=}$ die Differenz $d(v)=|v|_a-|v|_b$ für alle Präfixe $v$ von $w$.
\end{enumerate}

\begin{loesung}
\begin{enumerate}[a)]
  \item In jeder Satzform steht höchstens ein $S$, das jeweils ersetzt wird:
  \begin{align*}
    S &\Rightarrow aSb \Rightarrow aaSbb \Rightarrow aabb,\\
    S &\Rightarrow aSb \Rightarrow abSab \Rightarrow abab,\\
    S &\Rightarrow bSa \Rightarrow bbSaa \Rightarrow bbaa.
  \end{align*}

  \item Jede Anwendung von $S\to aSb$ oder $S\to bSa$ erzeugt genau zwei Buchstaben, am Ende steht einmal $S\to\varepsilon$. Alle Wörter in $L(G)$ haben daher gerade Länge. Wir zählen alle Ableitungen mit höchstens zwei \enquote{echten} Schritten auf:
  \begin{itemize}
    \item Länge 0: $S\Rightarrow\varepsilon$.
    \item Länge 2: $S\Rightarrow aSb\Rightarrow ab$ und $S\Rightarrow bSa\Rightarrow ba$.
    \item Länge 4: $aabb$, $abab$, $bbaa$ (siehe a)) sowie $S\Rightarrow bSa\Rightarrow baSba\Rightarrow baba$.
  \end{itemize}
  Also ist
  \[L_4=\{\varepsilon,\ ab,\ ba,\ aabb,\ abab,\ baba,\ bbaa\}.\]
  \textbf{$abba\notin L(G)$ und $baab\notin L(G)$:} Bei einem nichtleeren Wort ist der erste Schritt $S\to aSb$ oder $S\to bSa$. Die dabei erzeugten Buchstaben bleiben ganz außen stehen, weil alle weiteren Schritte nur das $S$ in der Mitte ersetzen. Jedes nichtleere Wort aus $L(G)$ beginnt also mit $a$ und endet mit $b$ oder umgekehrt. $abba$ beginnt und endet mit $a$, $baab$ beginnt und endet mit $b$. Beide Wörter liegen nicht in $L(G)$, obwohl sie gleich viele $a$ und $b$ enthalten.

  \item Invariante: Für jede Satzform $\alpha$ gilt $|\alpha|_a=|\alpha|_b$.
  \begin{itemize}
    \item Für $\alpha=S$ gilt $0=0$.
    \item Ein Schritt $\alpha\Rightarrow\beta$ ersetzt ein $S$ durch $aSb$, $bSa$ oder $\varepsilon$. Dabei steigen beide Anzahlen um $1$ oder bleiben gleich. Die Gleichheit bleibt also erhalten.
  \end{itemize}
  Per Induktion über die Anzahl der Schritte gilt die Invariante für alle Satzformen, insbesondere für alle $w\in L(G)$. \qed

  \item Für $x\in\{a,b\}^*$ sei $\ol{x}$ das Wort, das aus $x$ durch Vertauschen von $a$ und $b$ entsteht (z.\,B. $\ol{aab}=bba$). Dann gilt
  \[L(G)=\{x\,\ol{x}^R\mid x\in\{a,b\}^*\}.\]
  In Worten: Die Wörter aus $L(G)$ haben gerade Länge $2n$, und für alle $i\in\{1,\dots,n\}$ ist der $i$-te Buchstabe von vorne verschieden vom $i$-ten Buchstaben von hinten. Probe an $L_4$: $x=ab$ liefert $\ol{x}=ba$, $\ol{x}^R=ab$, also $abab$. $x=aa$ liefert $aabb$.

  \emph{$\subseteq$:} Alle Satzformen haben die Gestalt $x\,S\,\ol{x}^R$ oder $x\,\ol{x}^R$. Für $S=\varepsilon\,S\,\varepsilon$ stimmt das. Aus $x\,S\,\ol{x}^R$ wird mit $S\to aSb$ die Satzform $xa\,S\,b\,\ol{x}^R=(xa)\,S\,\ol{(xa)}^R$, denn $\ol{xa}=\ol{x}\,b$ und damit $\ol{xa}^R=b\,\ol{x}^R$. Für $S\to bSa$ geht es genauso, und $S\to\varepsilon$ liefert $x\,\ol{x}^R$.

  \emph{$\supseteq$:} Für $x=x_1\cdots x_n$ wendet man nacheinander für $i=1,\dots,n$ die Regel $S\to x_i\,S\,\ol{x_i}$ an und zum Schluss $S\to\varepsilon$.

  Anschaulich schreibt jede Regel vorne einen Buchstaben und an der gespiegelten Position hinten den jeweils anderen Buchstaben.

  \item Wir ergänzen die Regel $S\to SS$, also
  \[P':\qquad S\to aSb\mid bSa\mid SS\mid\varepsilon.\]
  \emph{$L(G')\subseteq L_{=}$:} Der Beweis aus c) funktioniert weiterhin, denn $S\to SS$ erzeugt keine Terminalsymbole.

  \emph{$L_{=}\subseteq L(G')$:} Wir zeigen per starker Induktion über $|w|$, dass jedes $w\in L_{=}$ aus $S$ ableitbar ist.
  \begin{itemize}
    \item $w=\varepsilon$: $S\Rightarrow\varepsilon$.
    \item Erster und letzter Buchstabe verschieden, also $w=aub$ oder $w=bua$: Dann ist auch $u\in L_{=}$ und $|u|<|w|$. Nach Induktionsvoraussetzung gilt $S\Rightarrow^* u$, also $S\Rightarrow aSb\Rightarrow^* aub$ (bzw.\ mit $bSa$).
    \item Erster und letzter Buchstabe gleich, etwa $w=aua$ (der Fall $b$ geht genauso): Für das Präfix aus dem ersten Buchstaben ist $d(a)=1$. Für das Präfix $au$ (alles außer dem letzten Buchstaben) ist $d(au)=d(w)-1=-1$. Da sich $d$ pro Buchstabe um genau $\pm1$ ändert, muss $d$ dazwischen den Wert $0$ annehmen. Es gibt also eine Zerlegung $w=vv'$ mit nichtleeren Wörtern $v,v'$ und $d(v)=0$. Wegen $d(w)=0$ ist auch $d(v')=0$, also $v,v'\in L_{=}$. Beide sind kürzer als $w$, nach Induktionsvoraussetzung ist $S\Rightarrow^* v$ und $S\Rightarrow^* v'$ und damit
    \[S\Rightarrow SS\Rightarrow^* vv'=w.\]
  \end{itemize}
  Beispiel: $abba=ab\cdot ba$, also $S\Rightarrow SS\Rightarrow aSbS\Rightarrow abS\Rightarrow abbSa\Rightarrow abba$. \qed

  (Eine andere richtige Lösung ist $S\to aSbS\mid bSaS\mid\varepsilon$. Dafür muss man aber alle drei Regeln ändern.)
\end{enumerate}
\end{loesung}

\aufgabetitel{$12$}{Welche Sprache erzeugt die Grammatik?}\\
Bestimmen Sie für jede der folgenden Grammatiken die erzeugte Sprache $L(G)$. Geben Sie sie in Mengenschreibweise an oder, wo das nicht sinnvoll geht, durch eine präzise Beschreibung in Worten. Begründen Sie Ihre Antwort.\\
\emph{Tipp:} Leiten Sie zuerst einige kurze Wörter ab und stellen Sie dann eine Vermutung auf.
\begin{enumerate}[a)]
  \item $S\to aS\mid b$
  \item $S\to aSbb\mid\varepsilon$
  \item $S\to aSb\mid aS\mid\varepsilon$
  \item $S\to AB,\quad A\to aA\mid\varepsilon,\quad B\to bB\mid\varepsilon$
  \item $S\to aSc\mid T,\quad T\to bTc\mid\varepsilon$
  \item $S\to aSa\mid bSb\mid a\mid b\mid\varepsilon$
  \item $S\to 0S\mid 1A\mid 1,\quad A\to 0A\mid 1S\mid 0\qquad$ (mit $\Sigma=\{0,1\}$)
  \item $S\to SS\mid\tm{(}S\tm{)}\mid\varepsilon\qquad$ (mit $\Sigma=\{\tm{(},\tm{)}\}$)
\end{enumerate}

\begin{loesung}
\begin{enumerate}[a)]
  \item Kurze Wörter: $b$, $ab$, $aab$, \ldots
  \[L(G)=\{a^nb\mid n\geq 0\}.\]
  Die Satzformen sind genau $a^nS$ und $a^nb$ mit $n\geq0$. Mit $S\to aS$ wächst der $a$-Block, mit $S\to b$ endet die Ableitung. Umgekehrt erhält man jedes $a^nb$ durch $n$-mal $S\to aS$ und einmal $S\to b$.

  \item Kurze Wörter: $\varepsilon$, $abb$, $aabbbb$, \ldots
  \[L(G)=\{a^nb^{2n}\mid n\geq 0\}.\]
  Hier der Beweis ausführlich als Muster für die anderen Teilaufgaben:

  \emph{$\supseteq$:} Für $n\geq0$ ist
  \[S\Rightarrow aSbb\Rightarrow aaSbbbb\Rightarrow\cdots\Rightarrow a^nSb^{2n}\Rightarrow a^nb^{2n}\]
  ($n$-mal $S\to aSbb$, dann $S\to\varepsilon$).

  \emph{$\subseteq$:} Invariante: Jede Satzform hat die Gestalt $a^nSb^{2n}$ oder $a^nb^{2n}$ mit $n\geq0$. Für $S=a^0Sb^0$ stimmt das. Eine Satzform $a^nb^{2n}$ enthält keine Variable mehr, dort geht es nicht weiter. Aus $a^nSb^{2n}$ entsteht $a^{n+1}Sb^{2(n+1)}$ oder $a^nb^{2n}$, beide wieder von der richtigen Gestalt. Die Satzformen ohne Variable sind also genau die Wörter $a^nb^{2n}$. \qed

  \item Kurze Wörter: $\varepsilon$, $a$, $aa$, $ab$, $aaa$, $aab$, \ldots
  \[L(G)=\{a^nb^m\mid n\geq m\geq 0\}.\]
  Jedes $b$ entsteht zusammen mit einem $a$ (Regel $S\to aSb$), zusätzliche $a$ liefert $S\to aS$. Da alle $a$ links und alle $b$ rechts von $S$ entstehen, stehen alle $a$ vor allen $b$.

  \emph{$\subseteq$:} Die Satzformen haben die Gestalt $a^nSb^m$ oder $a^nb^m$ mit $n\geq m$. Die Regel $S\to aSb$ erhöht beide Exponenten um $1$, $S\to aS$ nur $n$.

  \emph{$\supseteq$:} Für $n\geq m$ wendet man $m$-mal $S\to aSb$ an, dann $(n-m)$-mal $S\to aS$ und schließlich $S\to\varepsilon$:
  \[S\Rightarrow^* a^mSb^m\Rightarrow^* a^nSb^m\Rightarrow a^nb^m.\]

  \item $A$ erzeugt genau die Wörter $a^m$ mit $m\geq0$, $B$ genau die Wörter $b^n$ mit $n\geq0$ (wie in a), nur mit $\varepsilon$ statt $b$ als Abschluss). Damit ist
  \[L(G)=\{a^mb^n\mid m,n\geq 0\}.\]
  Im Unterschied zu c) sind die beiden Anzahlen hier völlig unabhängig: $A$ und $B$ werden getrennt voneinander abgeleitet. In c) koppelt die Regel $S\to aSb$ die beiden Anzahlen.

  \item Kurze Wörter: $\varepsilon$, $ac$, $bc$, $aacc$, $abcc$, $bbcc$, \ldots
  \[L(G)=\{a^nb^mc^{n+m}\mid n,m\geq 0\}.\]
  Die Ableitung hat zwei Phasen. Zuerst erzeugt $S\to aSc$ die Satzform $a^nSc^n$. Mit $S\to T$ wechselt man in die zweite Phase, in der $T\to bTc$ die Satzform $a^nb^mTc^mc^n$ erzeugt. Zum Schluss kommt $T\to\varepsilon$. Jedes $a$ und jedes $b$ bekommt also sein eigenes $c$. Da es von $T$ keinen Weg zurück zu $S$ gibt, kann nach einem $b$ kein $a$ mehr kommen. Umgekehrt erhält man jedes $a^nb^mc^{n+m}$ mit genau diesem Schema.

  \item Kurze Wörter: $\varepsilon$, $a$, $b$, $aa$, $bb$, $aaa$, $aba$, $bab$, $bbb$, \ldots Die erzeugte Sprache ist die Menge der \textbf{Palindrome} über $\{a,b\}$:
  \[L(G)=\{w\in\{a,b\}^*\mid w=w^R\}.\]
  \emph{$\subseteq$:} Die Satzformen haben die Gestalt $xSx^R$ oder $x\,c\,x^R$ mit $c\in\{\varepsilon,a,b\}$, denn $S\to aSa$ macht aus $xSx^R$ die Satzform $(xa)S(xa)^R$. Jedes Wort $x\,c\,x^R$ ist ein Palindrom.

  \emph{$\supseteq$:} Ist $w=w^R$ und $n=\lfloor|w|/2\rfloor$, so ist $w=x\,c\,x^R$, wobei $x$ aus den ersten $n$ Buchstaben besteht und $c\in\{\varepsilon,a,b\}$ ist. Man erzeugt $x$ von außen nach innen mit $S\to aSa$ bzw.\ $S\to bSb$ und schließt mit $S\to c$ ab.

  \item Kurze Wörter: $1$, $01$, $10$, $001$, $010$, $100$, $111$, \ldots\ Vermutung:
  \[L(G)=\{w\in\{0,1\}^*\mid |w|_1\text{ ist ungerade}\}.\]
  \emph{Idee:} Die Variable \enquote{merkt sich}, ob bisher eine gerade ($S$) oder ungerade ($A$) Anzahl von Einsen erzeugt wurde. Eine $0$ ändert nichts, eine $1$ wechselt die Variable.

  \emph{$\subseteq$:} Invariante: Jede Satzform ist entweder $uS$ mit $|u|_1$ gerade oder $uA$ mit $|u|_1$ ungerade oder ein Wort $u$ ohne Variable mit $|u|_1$ ungerade. Das prüft man für alle Regeln einzeln nach. Zum Beispiel macht $A\to 1S$ aus einer ungeraden Anzahl eine gerade, $S\to 1$ macht aus einer geraden Anzahl eine ungerade und beendet die Ableitung, und $A\to 0$ beendet die Ableitung bei ungerader Anzahl.

  \emph{$\supseteq$:} Sei $w=w_1\cdots w_k$ mit ungerader Anzahl von Einsen (insbesondere $k\geq1$). Man erzeugt $w_1,\dots,w_{k-1}$ von links nach rechts und wählt dabei jeweils die Regel, die zur neuen Parität passt. Für den letzten Buchstaben gilt: Ist $w_k=1$, so enthält $w_1\cdots w_{k-1}$ eine gerade Anzahl von Einsen, die Variable ist $S$, und man schließt mit $S\to1$ ab. Ist $w_k=0$, so ist die Anzahl ungerade, die Variable ist $A$, und man schließt mit $A\to0$ ab. Beispiel:
  \[S\Rightarrow 1A\Rightarrow 10A\Rightarrow 101S\Rightarrow 1011.\]
  Diese Idee, in der Variablen eine endliche Information über das bisher erzeugte Wort zu speichern, begegnet uns bei den endlichen Automaten wieder.

  \item Kurze Wörter: $\varepsilon$, $\tm{()}$, $\tm{()()}$, $\tm{(())}$, $\tm{()()()}$, $\tm{(()())}$, \ldots\ Erzeugt werden genau die \textbf{korrekt geklammerten Ausdrücke}:
  \[L(G)=\bigl\{w\in\{\tm{(},\tm{)}\}^*\ \bigm|\ |w|_{\tm{(}}=|w|_{\tm{)}}\text{ und für jedes Präfix $v$ von $w$ gilt }|v|_{\tm{(}}\geq|v|_{\tm{)}}\bigr\}.\]
  In Worten: Man schließt nie mehr Klammern, als man bisher geöffnet hat, und am Ende sind alle Klammern geschlossen. Zum Beispiel sind $\tm{(()())}$ und $\tm{()(())}$ enthalten, $\tm{)(}$, $\tm{())}$ und $\tm{(()}$ dagegen nicht. Wir nennen diese Menge $D$.

  \emph{$\subseteq$:} Wir betrachten in einer Satzform nur die Klammern und ignorieren die Variablen. Invariante: Die Satzform ist ausgeglichen, und jedes Präfix hat mindestens so viele öffnende wie schließende Klammern. Die Regeln $S\to SS$ und $S\to\varepsilon$ ändern an den Klammern nichts. $S\to\tm{(}S\tm{)}$ fügt ein zusammengehöriges Paar ein: Präfixe, die das neue $\tm{(}$, aber nicht das neue $\tm{)}$ enthalten, bekommen eine öffnende Klammer mehr, alle anderen Präfixe behalten ihre Bilanz.

  \emph{$\supseteq$:} Starke Induktion über $|w|$. Für $w=\varepsilon$ gilt $S\Rightarrow\varepsilon$. Sei $w\in D$ nichtleer. Dann beginnt $w$ mit $\tm{(}$. Sei $v$ das kürzeste nichtleere Präfix von $w$ mit gleich vielen öffnenden und schließenden Klammern. Dann ist $v=\tm{(}u\tm{)}$ mit $u\in D$ (innerhalb von $v$ sinkt die Bilanz erst am Ende auf $0$), und $w=vw'$ mit $w'\in D$. Nach Induktionsvoraussetzung gilt
  \[S\Rightarrow SS\Rightarrow\tm{(}S\tm{)}S\Rightarrow^*\tm{(}u\tm{)}w'=w.\]
  (Das ist derselbe Präfix-Trick wie in Aufgabe 1 e).)
\end{enumerate}
\end{loesung}
